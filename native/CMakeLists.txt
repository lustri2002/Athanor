cmake_minimum_required(VERSION 3.24)
# This entry point also supports configuring native/ directly.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" ATHANOR_VERSION)
string(STRIP "${ATHANOR_VERSION}" ATHANOR_VERSION)
if(NOT ATHANOR_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
 message(FATAL_ERROR "VERSION must contain a numeric major.minor.patch version")
endif()
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
 get_filename_component(athanor_version_file "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" ABSOLUTE)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${athanor_version_file}")
endif()
project(Athanor VERSION "${ATHANOR_VERSION}" LANGUAGES C CXX)
string(REGEX REPLACE "\\.0$" "" ATHANOR_RELEASE_VERSION "${PROJECT_VERSION}")
set(ATHANOR_APPLICATION_VERSION "${ATHANOR_RELEASE_VERSION}-alpha")
set(athanor_architecture "${CMAKE_SYSTEM_PROCESSOR}")
if(CMAKE_CXX_COMPILER_ARCHITECTURE_ID)
 set(athanor_architecture "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
endif()
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
 list(LENGTH CMAKE_OSX_ARCHITECTURES athanor_architecture_count)
 if(athanor_architecture_count EQUAL 1)
  set(athanor_architecture "${CMAKE_OSX_ARCHITECTURES}")
 else()
  set(athanor_architecture "universal")
 endif()
endif()
string(TOLOWER "${athanor_architecture}" athanor_architecture)
if(athanor_architecture MATCHES "^(x86_64|amd64|x64)$")
 set(athanor_architecture "x64")
elseif(athanor_architecture MATCHES "^(aarch64|arm64)$")
 set(athanor_architecture "arm64")
elseif(athanor_architecture MATCHES "^(i[3-6]86|x86)$")
 set(athanor_architecture "x86")
endif()
if(WIN32)
 set(athanor_platform "Windows")
 set(athanor_metadata_destination ".")
elseif(APPLE)
 set(athanor_platform "macOS")
 set(athanor_metadata_destination "Athanor.app/Contents/Resources")
else()
 set(athanor_platform "Linux")
 set(athanor_metadata_destination "share/athanor")
endif()
configure_file(version.h.in "${CMAKE_CURRENT_BINARY_DIR}/version.h" @ONLY)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/athanor-version.json" CONTENT
 "{\n  \"version\": \"${PROJECT_VERSION}\",\n  \"application_version\": \"${ATHANOR_APPLICATION_VERSION}\",\n  \"release_version\": \"${ATHANOR_RELEASE_VERSION}\",\n  \"platform\": \"${athanor_platform}\",\n  \"architecture\": \"${athanor_architecture}\",\n  \"asset_basename\": \"Athanor-Alpha-${ATHANOR_RELEASE_VERSION}-${athanor_platform}-${athanor_architecture}\"\n}\n")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ATHANOR_BUNDLE_TOOLS "Copy conversion tools into the install tree" OFF)
set(ATHANOR_TOOLS_ROOT "" CACHE PATH "Directory containing ffmpeg, ffprobe and avifenc")
set(ATHANOR_MUPDF_ROOT "" CACHE PATH "MuPDF include and library prefix")
option(ATHANOR_MANUAL_MOC "Use existing generated metaobjects in a managed build" OFF)
if(ATHANOR_MANUAL_MOC)
 set(CMAKE_AUTOMOC OFF)
else()
 set(CMAKE_AUTOMOC ON)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Concurrent Svg ShaderTools Network)
set(ATHANOR_CODEC_ROOT "" CACHE PATH "Native codec development files")
qt_add_executable(Athanor WIN32 main.cpp updater.cpp updater.h controller.cpp controller.h childprocess.cpp childprocess.h conversion.cpp conversion.h platform.cpp platform.h paths.cpp desktopintegration.cpp desktopintegration.h unixupdate.cpp unixupdate.h pdf_bridge.c pdf_bridge.h ui_test.cpp ui_test.h platform_test.cpp)
target_include_directories(Athanor PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_sources(Athanor PRIVATE process_test.cpp process_test.h)
target_sources(Athanor PRIVATE inputselection.h)
if(ATHANOR_MANUAL_MOC)
 target_sources(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/generated/moc_updater.cpp" "${ATHANOR_CODEC_ROOT}/generated/moc_controller.cpp" "${ATHANOR_CODEC_ROOT}/generated/moc_childprocess.cpp")
endif()
qt_add_resources(Athanor "app_qml" PREFIX "/qt/qml/Athanor" FILES Main.qml Quick.qml Preview.qml ActionButton.qml Card.qml Field.qml Spin.qml Check.qml Radio.qml SizeChoice.qml SizeAmount.qml Checker.qml Theme.qml Glyph.qml Input.qml QueuePanel.qml Hint.qml Formats.qml FormatSelect.qml FormatGuide.qml ConversionPrompts.qml qmldir)
qt_add_shaders(Athanor "reveal_shader" PREFIX "/shaders" FILES reveal.frag rounded-mask.frag checker.frag)
target_link_libraries(Athanor PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Svg Qt6::Network)
if(UNIX AND NOT APPLE)
 find_package(Qt6 6.8 REQUIRED COMPONENTS DBus)
 target_link_libraries(Athanor PRIVATE Qt6::DBus)
endif()
# Keep the original managed Windows codec SDK supported.
if(WIN32 AND ATHANOR_CODEC_ROOT)
 target_include_directories(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/mupdf/include" "${ATHANOR_CODEC_ROOT}/avif/include" "${ATHANOR_CODEC_ROOT}/webp/Library/include" "${ATHANOR_CODEC_ROOT}/heif/include")
 target_link_libraries(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/mupdf.lib" "${ATHANOR_CODEC_ROOT}/avif.lib" "${ATHANOR_CODEC_ROOT}/heif.lib" "${ATHANOR_CODEC_ROOT}/webp/Library/lib/libwebp.lib" "${ATHANOR_CODEC_ROOT}/webp/Library/lib/libwebpmux.lib" "${ATHANOR_CODEC_ROOT}/webp/Library/lib/libwebpdemux.lib")
else()
 include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/Codecs.cmake)
 target_link_libraries(Athanor PRIVATE AthanorCodecs)
endif()
if(WIN32)
 if(MSVC)
  target_compile_options(Athanor PRIVATE /utf-8)
 endif()
 target_link_libraries(Athanor PRIVATE shell32 ole32 dwmapi)
 configure_file(athanor.rc.in "${CMAKE_CURRENT_BINARY_DIR}/athanor.rc" @ONLY)
 target_sources(Athanor PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/athanor.rc")
elseif(APPLE)
 enable_language(OBJCXX)
 target_sources(Athanor PRIVATE macos.mm)
 target_link_libraries(Athanor PRIVATE "-framework AppKit")
 set_target_properties(Athanor PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.SixFawn253.Athanor" MACOSX_BUNDLE_BUNDLE_NAME "Athanor" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/Info.plist.in")
endif()
file(GLOB app_assets CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../assets/*")
qt_add_resources(Athanor "app_assets" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}/.." FILES ${app_assets})
if(APPLE)
 set(athanor_tool_destination "Athanor.app/Contents/Resources/tools")
else()
 set(athanor_tool_destination "tools")
 if(UNIX)
  set(athanor_tool_destination "bin")
 endif()
endif()
if(ATHANOR_BUNDLE_TOOLS)
 foreach(tool ffmpeg ffprobe avifenc)
  find_program(ATHANOR_${tool} NAMES ${tool} HINTS "${ATHANOR_TOOLS_ROOT}" REQUIRED)
  install(PROGRAMS "${ATHANOR_${tool}}" DESTINATION "${athanor_tool_destination}")
 endforeach()
endif()
if(UNIX AND NOT APPLE)
 install(TARGETS Athanor RUNTIME DESTINATION bin)
else()
 install(TARGETS Athanor BUNDLE DESTINATION . RUNTIME DESTINATION .)
endif()
if(WIN32)
 install(FILES qt.conf DESTINATION .)
endif()
if(UNIX AND NOT APPLE)
 install(FILES ../packaging/io.github.SixFawn253.Athanor.desktop DESTINATION share/applications)
 install(FILES ../assets/logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME io.github.SixFawn253.Athanor.png)
endif()
install(DIRECTORY ../licenses/ DESTINATION licenses)
install(FILES ../THIRD-PARTY-NOTICES.md DESTINATION .)
install(FILES "${CMAKE_BINARY_DIR}/athanor-version.json" DESTINATION "${athanor_metadata_destination}")
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME version COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/version.py" "$<TARGET_FILE:Athanor>" "${CMAKE_BINARY_DIR}/athanor-version.json")
 set_tests_properties(version PROPERTIES TIMEOUT 30 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 add_test(NAME conversions COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/smoke.py" "$<TARGET_FILE:Athanor>")
 add_test(NAME updater COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/updater.py" "$<TARGET_FILE:Athanor>")
 add_test(NAME update_download COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/update_download.py" "$<TARGET_FILE:Athanor>")
 set_tests_properties(update_download PROPERTIES TIMEOUT 60 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 add_test(NAME cli COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/cli.py" "$<TARGET_FILE:Athanor>")
 set_tests_properties(cli PROPERTIES TIMEOUT 90 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 add_test(NAME media_regressions COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/media_regressions.py" "$<TARGET_FILE:Athanor>")
 set_tests_properties(media_regressions PROPERTIES TIMEOUT 180 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 add_test(NAME preview_regressions COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../tests/preview_regressions.py" "$<TARGET_FILE:Athanor>")
 set_tests_properties(preview_regressions PROPERTIES TIMEOUT 180 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 add_test(NAME processes COMMAND Athanor --process-test)
 set_tests_properties(processes PROPERTIES TIMEOUT 60 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 set_tests_properties(updater PROPERTIES TIMEOUT 90)
 add_test(NAME platform COMMAND Athanor --platform-test)
 set_tests_properties(platform PROPERTIES TIMEOUT 60 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
 set_tests_properties(conversions PROPERTIES TIMEOUT 300 ENVIRONMENT "ATHANOR_TEST=1;QT_QPA_PLATFORM=offscreen")
endif()
