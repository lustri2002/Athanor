cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ATHANOR_VERSION)
string(STRIP "${ATHANOR_VERSION}" ATHANOR_VERSION)
if(NOT ATHANOR_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
 message(FATAL_ERROR "VERSION must contain a numeric major.minor.patch version")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(AthanorNative VERSION "${ATHANOR_VERSION}" LANGUAGES C CXX)
include(CTest)
add_subdirectory(native)
